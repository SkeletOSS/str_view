31/* SV_ATTRIBUTE_PURE has no side effects when given the same arguments with the
32 same data, producing the same return value. A SV_Str_view points to char
33 const * data which may change in between SV_Str_view calls.
34 SV_ATTRIBUTE_CONST applies only where no pointers are accessed or
35 dereferenced. Other attributes provide more string safety and opportunities
36 to optimize. Credit Harith on Code Review Stack Exchange. */

51# define SV_ATTRIBUTE_CONST __attribute__((const))
52# else

60 __attribute__((null_terminated_string_arg(__VA_ARGS__)))

64# endif
65# else
68# define SV_ATTRIBUTE_PURE
71# define SV_ATTRIBUTE_CONST
